import numpy as npimport pandas as pd # EVOLVE-BLOCK-STARTdef run_experiment(bars: pd.DataFrame) -> np.ndarray:    close = bars["close"]    ret_20 = close.pct_change(20)    vol_60 = close.pct_change().rolling(60).std()    target = 0.02 / vol_60.clip(lower=0.005)    position = (np.sign(ret_20) * target).clip(-1, 1)    return position.fillna(0).to_numpy()# EVOLVE-BLOCK-END
